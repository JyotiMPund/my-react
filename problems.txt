//Write a program to find count of duplicate number from an array
CODE:
let arrNum = [1, 2, 3, 4, 4, 5, 5, 5];

// Count occurrences
let countObj = {};
arrNum.forEach(num => countObj[num] = (countObj[num] || 0) + 1);

// Display duplicates and counts
for (let num in countObj) {
    if (countObj[num] > 1) {
        console.log(`Number: ${num}, Count: ${countObj[num]}`);
    }
}
